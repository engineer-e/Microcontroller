#define F_CPU 16000000UL

#include <avr/io.h>
#include <util/delay.h>

#define BAUD_RATE 9600UL

// Check whether UART data has been received
#define uartReady (UCSRA & (1 << RXC))

// Initialize UART
void uartInit(unsigned long baud)
{
    unsigned int ubrr;

    // Calculate baud rate register value
    ubrr = (F_CPU / (16UL * baud)) - 1;

    UBRRH = (unsigned char)(ubrr >> 8);
    UBRRL = (unsigned char)ubrr;

    // Enable UART receiver and transmitter
    UCSRB = (1 << RXEN) | (1 << TXEN);

    // Asynchronous mode, 8 data bits, no parity, 1 stop bit
    UCSRC = (1 << URSEL) | (1 << UCSZ1) | (1 << UCSZ0);
}

// Transmit one character
void uartTransmit(unsigned char data)
{
    // Wait until transmit buffer is empty
    while (!(UCSRA & (1 << UDRE)))
    {
    }

    UDR = data;
}

// Receive one character
unsigned char uartReceive(void)
{
    // Wait until data is received
    while (!(UCSRA & (1 << RXC)))
    {
    }

    return UDR;
}

// Transmit a string
void sendText(const char *txt)
{
    while (*txt != '\0')
    {
        uartTransmit((unsigned char)*txt);
        txt++;
    }
}

int main(void)
{
    unsigned char tmp;
    unsigned char i = 0;
    char txt[16];

    // Configure PORTC as output
    DDRC = 0xFF;

    // Initialize UART at 9600 baud
    uartInit(BAUD_RATE);

    // Send startup message
    sendText("ATMEGA32 AVR UART Example 1 = Gobal Krishnan V & Microchip & Chatgpt\r\n");

    while (1)
    {
        // Check whether a character has arrived
        if (uartReady)
        {
            tmp = uartReceive();

            // Display received character on PORTC
            PORTC = tmp;

            // Check for Enter key (carriage return)
            if (tmp == '\r' || tmp == '\n')
            {
                if (i > 0)
                {
                    // Terminate the received string
                    txt[i] = '\0';

                    // Send received string back to terminal
                    sendText(txt);
                    sendText("\r\n");

                    // Clear buffer index for the next string
                    i = 0;
                }
            }
            else
            {
                // Store character only if space is available
                if (i < sizeof(txt) - 1)
                {
                    txt[i++] = (char)tmp;
                }
            }
        }
    }
}
